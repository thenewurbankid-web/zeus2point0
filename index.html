Strategy API reference 
function onCandle(ctx) { ... }   // called each time a trigger-TF candle closes

ctx.symbol, ctx.tfName, ctx.price, ctx.time
ctx.tf('15m')        -> closed candles [{t,o,h,l,c,v}] (last 400), default = trigger TF
ctx.closes(tf) / highs(tf) / lows(tf) / volumes(tf)  -> number[]
ctx.ind.sma(v,p) ema(v,p) rsi(v,p) atr(candles,p) vwap(candles)
ctx.ind.macd(v,12,26,9) -> {macd,signal,hist}   bb(v,20,2) -> {upper,mid,lower}
ctx.ind.supertrend(candles,10,3) -> {line,dir}   adx(candles,14) -> {adx,pdi,mdi}
ctx.ind.highest(v,p) lowest(v,p)
ctx.last(arr, n=0)   ctx.crossOver(a,b)   ctx.crossUnder(a,b)   (a/b = arrays or numbers)
ctx.trend('1h') -> score -5..+5     ctx.trendDetail('1h')     ctx.confluence() -> -5..+5
ctx.position -> null | {dir (1 long, -1 short), lev, entry, qty, sl, tp, liq, pnlPct, openedAt}
ctx.market -> 'spot' | 'perp'   ctx.leverage
ctx.short({sizePct, slPct, tpPct}) -> open a short (perp strategies only)
ctx.state    -> object that survives between calls
ctx.buy({slPct, tpPct, sl, tp, trail, sizePct})   // all optional; overrides Risk settings
ctx.exit(reason)     ctx.log(...msg)
